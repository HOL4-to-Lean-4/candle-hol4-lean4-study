    (∀(s :term) (s' :term).
       MEM ((s,s') :term # term) env ⇒ ∃(x :mlstring) (ty :type). s = Var x ty ∧ s' = Var x (TYPE_SUBST tyin ty)) ∧
    IS_RESULT (INST_CORE env tyin tm) ⇒
    term_ok sig (RESULT (INST_CORE env tyin tm))

=== holSyntaxExtra.term_ok_INST

    term_ok sig tm ∧ EVERY (type_ok (tysof sig)) (MAP (FST :type # type -> type) tyin) ⇒ term_ok sig (INST tyin tm)

=== holSyntaxExtra.term_ok_raconv

    RACONV env tp ⇒
    ∀(sig :(mlstring |-> num) # (mlstring |-> type)).
      EVERY
        (λ(((s :term),(s' :term)) :term # term).
           welltyped s ∧ welltyped s' ∧ typeof s = typeof s' ∧ type_ok (tysof sig) (typeof s)) env ⇒
      term_ok sig (FST tp) ∧ welltyped (SND tp) ⇒
      term_ok sig (SND tp)

=== holSyntaxExtra.term_ok_aconv
⊢ ∀(sig :(mlstring |-> num) # (mlstring |-> type)) (t1 :term) (t2 :term).
    ACONV t1 t2 ∧ term_ok sig t1 ∧ welltyped t2 ⇒ term_ok sig t2

=== holSyntaxExtra.bind_not_free

=== holSyntaxExtra.MAP_db_FILTER_neq
